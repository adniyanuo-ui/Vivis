function(header_library target directory)
  add_library(${target} INTERFACE IMPORTED GLOBAL)
  set_target_properties(${target} PROPERTIES INTERFACE_INCLUDE_DIRECTORIES "${directory}")
endfunction()

# CCCL bundled with CUDA13 no longer exports thrust::pair. Deduce the
# minmax_element return type; keep the SDK checkout and numeric algorithm intact.
foreach(BASE animator_cuda multitrack_animator_cuda)
  set(ANIMATOR_SOURCE "${SDK}/audio2face-sdk/source/audio2face-core/${BASE}.cu")
  file(READ "${ANIMATOR_SOURCE}" ANIMATOR_TEXT)
  set(OLD_PAIR "thrust::pair<const thrust::device_ptr<float>, const thrust::device_ptr<float>> minmax =")
  string(FIND "${ANIMATOR_TEXT}" "${OLD_PAIR}" PAIR_OFFSET)
  if(PAIR_OFFSET EQUAL -1)
    message(FATAL_ERROR "Pinned SDK compatibility pattern no longer matches")
  endif()
  string(REPLACE "${OLD_PAIR}" "auto minmax =" ANIMATOR_TEXT "${ANIMATOR_TEXT}")
  set(COMPAT_PATH "${CMAKE_BINARY_DIR}/${BASE}_compat.cu")
  set(EXISTING_TEXT "")
  if(EXISTS "${COMPAT_PATH}")
    file(READ "${COMPAT_PATH}" EXISTING_TEXT)
  endif()
  if(NOT "${EXISTING_TEXT}" STREQUAL "${ANIMATOR_TEXT}")
    file(WRITE "${COMPAT_PATH}" "${ANIMATOR_TEXT}")
  endif()
  get_target_property(FACE_SOURCES audio2face-core SOURCES)
  list(FIND FACE_SOURCES "${ANIMATOR_SOURCE}" SOURCE_OFFSET)
  if(SOURCE_OFFSET EQUAL -1)
    message(FATAL_ERROR "Original CUDA source not found in SDK target")
  endif()
  list(REMOVE_ITEM FACE_SOURCES "${ANIMATOR_SOURCE}")
  set_target_properties(audio2face-core PROPERTIES SOURCES "${FACE_SOURCES}")
  target_sources(audio2face-core PRIVATE "${CMAKE_BINARY_DIR}/${BASE}_compat.cu")
endforeach()
