cmake_minimum_required(VERSION 3.16)
project(vivis_seeta_probe LANGUAGES CXX)
set(SEETA_PREFIX "" CACHE PATH "Local SeetaFace6 install prefix")
if(NOT IS_DIRECTORY "${SEETA_PREFIX}/include")
  message(FATAL_ERROR "Set SEETA_PREFIX to the isolated SeetaFace6 build prefix")
endif()
add_library(vivis_seeta_probe SHARED probe.cpp)
target_compile_features(vivis_seeta_probe PRIVATE cxx_std_11)
target_include_directories(vivis_seeta_probe PRIVATE "${SEETA_PREFIX}/include")
foreach(name SeetaFaceDetector600 SeetaFaceLandmarker600 SeetaFaceRecognizer610 SeetaQualityAssessor300 SeetaAgePredictor600)
  find_library(lib_${name} NAMES ${name} PATHS "${SEETA_PREFIX}/lib" NO_DEFAULT_PATH REQUIRED)
  target_link_libraries(vivis_seeta_probe PRIVATE "${lib_${name}}")
endforeach()
# Local diagnostic only; no /usr/local install or global loader modification.
set_target_properties(vivis_seeta_probe PROPERTIES BUILD_RPATH "${SEETA_PREFIX}/lib")
target_link_options(vivis_seeta_probe PRIVATE "-Wl,--disable-new-dtags")
add_executable(seeta_quality_test quality_test.cpp)
target_compile_features(seeta_quality_test PRIVATE cxx_std_11)
target_include_directories(seeta_quality_test PRIVATE "${SEETA_PREFIX}/include")
target_link_libraries(seeta_quality_test PRIVATE "${lib_SeetaQualityAssessor300}")
set_target_properties(seeta_quality_test PROPERTIES BUILD_RPATH "${SEETA_PREFIX}/lib")
target_link_options(seeta_quality_test PRIVATE "-Wl,--disable-new-dtags")
