cmake_minimum_required(VERSION 3.24)
project(eva_a2f_environment LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
get_filename_component(EVA_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
add_subdirectory("${EVA_ROOT}/third_party/Audio2Face-3D-SDK" sdk)
add_executable(eva-a2f-smoke ../a2f_smoke.cpp)
target_compile_definitions(eva-a2f-smoke PRIVATE TEST_DATA_DIR="${EVA_ROOT}/" _GLIBCXX_USE_CXX11_ABI=0)
target_include_directories(eva-a2f-smoke PRIVATE "${EVA_ROOT}/third_party/Audio2Face-3D-SDK/_deps/target-deps/audiofile/include")
target_link_libraries(eva-a2f-smoke PRIVATE audio2x)

add_executable(eva-a2f-export ../a2f_export.cpp)
target_compile_definitions(eva-a2f-export PRIVATE TEST_DATA_DIR="${EVA_ROOT}/" _GLIBCXX_USE_CXX11_ABI=0)
target_include_directories(eva-a2f-export PRIVATE "${EVA_ROOT}/third_party/Audio2Face-3D-SDK/_deps/target-deps/audiofile/include")
target_include_directories(eva-a2f-export PRIVATE "${EVA_ROOT}/tools/cuda-12.9/include" "${EVA_ROOT}/third_party/Audio2Face-3D-SDK/_deps/target-deps/nlohmann_json/include")
target_link_libraries(eva-a2f-export PRIVATE audio2x "${EVA_ROOT}/tools/cuda-12.9/lib/libcudart.so")
add_executable(eva-a2f-solve ../a2f_solve_weights.cpp)
target_compile_definitions(eva-a2f-solve PRIVATE _GLIBCXX_USE_CXX11_ABI=0)
target_include_directories(eva-a2f-solve PRIVATE "${EVA_ROOT}/tools/cuda-12.9/include" "${EVA_ROOT}/third_party/Audio2Face-3D-SDK/_deps/target-deps/nlohmann_json/include")
target_link_libraries(eva-a2f-solve PRIVATE audio2x "${EVA_ROOT}/tools/cuda-12.9/lib/libcudart.so")
